\section{Теоретическое
введение}\label{ux442ux435ux43eux440ux435ux442ux438ux447ux435ux441ux43aux43eux435-ux432ux432ux435ux434ux435ux43dux438ux435}

\begin{frame}{Модель SIR}
\phantomsection\label{ux43cux43eux434ux435ux43bux44c-sir}
\begin{itemize}[<+->]
\tightlist
\item
  Классическая эпидемиологическая модель
\item
  Популяция делится на три группы:

  \begin{itemize}[<+->]
  \tightlist
  \item
    \textbf{S (Susceptible)} --- восприимчивые
  \item
    \textbf{I (Infectious)} --- инфицированные
  \item
    \textbf{R (Recovered)} --- выздоровевшие
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Система уравнений SIR}
\phantomsection\label{ux441ux438ux441ux442ux435ux43cux430-ux443ux440ux430ux432ux43dux435ux43dux438ux439-sir}
\[
\begin{cases}
\dfrac{dS}{dt} = -\beta \dfrac{S I}{N} \\[1em]
\dfrac{dI}{dt} = \beta \dfrac{S I}{N} - \gamma I \\[1em]
\dfrac{dR}{dt} = \gamma I
\end{cases}
\]

где: - ( N = S + I + R ) --- общая численность популяции - ( \beta ) ---
коэффициент заражения - ( \gamma ) --- коэффициент выздоровления
\end{frame}

\begin{frame}{Ключевые показатели}
\phantomsection\label{ux43aux43bux44eux447ux435ux432ux44bux435-ux43fux43eux43aux430ux437ux430ux442ux435ux43bux438}
\begin{itemize}[<+->]
\tightlist
\item
  \textbf{Базовое репродуктивное число}: ( R\_0 = \dfrac{\beta}{\gamma}
  )

  \begin{itemize}[<+->]
  \tightlist
  \item
    ( R\_0 \textgreater{} 1 ) --- эпидемия разрастается
  \item
    ( R\_0 \textless{} 1 ) --- эпидемия затухает
  \end{itemize}
\item
  \textbf{Порог коллективного иммунитета}: ( 1 - \dfrac{1}{R_0} )
\item
  \textbf{Эффективное репродуктивное число}: ( R\_e(t) = R\_0
  \cdot \frac{S(t)}{N} )
\end{itemize}
\end{frame}

\begin{frame}{Модель Лотки-Вольтерры}
\phantomsection\label{ux43cux43eux434ux435ux43bux44c-ux43bux43eux442ux43aux438-ux432ux43eux43bux44cux442ux435ux440ux440ux44b}
\begin{itemize}[<+->]
\tightlist
\item
  Модель «хищник-жертва»
\item
  Описывает взаимодействие двух биологических видов
\end{itemize}

\begin{block}{Система уравнений}
\phantomsection\label{ux441ux438ux441ux442ux435ux43cux430-ux443ux440ux430ux432ux43dux435ux43dux438ux439}
\[
\begin{cases}
\dfrac{dx}{dt} = \alpha x - \beta x y \\[1em]
\dfrac{dy}{dt} = \delta x y - \gamma y
\end{cases}
\]
\end{block}
\end{frame}

\begin{frame}{Параметры модели Лотки-Вольтерры}
\phantomsection\label{ux43fux430ux440ux430ux43cux435ux442ux440ux44b-ux43cux43eux434ux435ux43bux438-ux43bux43eux442ux43aux438-ux432ux43eux43bux44cux442ux435ux440ux440ux44b}
\begin{itemize}[<+->]
\tightlist
\item
  ( x ) --- численность жертв
\item
  ( y ) --- численность хищников
\item
  ( \alpha ) --- прирост жертв
\item
  ( \beta ) --- смертность жертв от хищников
\item
  ( \delta ) --- прирост хищников от жертв
\item
  ( \gamma ) --- смертность хищников
\end{itemize}
\end{frame}

\begin{frame}{Стационарная точка}
\phantomsection\label{ux441ux442ux430ux446ux438ux43eux43dux430ux440ux43dux430ux44f-ux442ux43eux447ux43aux430}
\begin{itemize}[<+->]
\tightlist
\item
  Нетривиальная точка сосуществования видов:
\end{itemize}

{[} x\^{}* = \frac{\gamma}{\delta},\quad y\^{}* = \frac{\alpha}{\beta}
{]}

\begin{itemize}[<+->]
\tightlist
\item
  Вокруг этой точки решения носят колебательный характер
\item
  Пик численности хищников отстаёт от пика численности жертв
\end{itemize}
\end{frame}
